### 1. 入力バリデーションの実装

そのままだと空文字でもやけに長いURLでもURLじゃなくても短縮キーを発行してしまうので、Lambdaのハンドラで入力バリデーションを実装する。

```typescript
const MAX_URL_LENGTH = 2048
const validateLongUrl = (value: unknown): string | null => {
  if (typeof value !== "string" || value === "") return "longUrl must be a non-empty string.";
  if (value.length > MAX_URL_LENGTH) return `longUrl must be at most ${MAX_URL_LENGTH} chars`;
  let parsed: URL;
  try {
    parsed = new URL(value);
  } catch {
    return "longUrl is not a valid URL";
  }

  if (parsed.protocol !== "https:") return "only https are allowed";

  return null;
}

export const handler = async (event: any) => {
  let body: any;
  try {
    body = JSON.parse(event.body ?? "");
  } catch {
    return { statusCode: 400, body: JSON.stringify({ error: "invalid JSON" }) };
  }

  const error = validateLongUrl(body?.longUrl);
  if (error) {
    return { statusCode: 400, body: JSON.stringify({ error }) };
  }

  const longUrl = body.longUrl as string;

  if (!longUrl) {
    return { statusCode: 400, body: JSON.stringify({ error: "longUrl is required" }) };
  }

  const shortKey = await putWithRetry(longUrl);

  return { statusCode: 200, body: JSON.stringify({ shortKey }) };
};
```

### 2. Terraformちょっと触ってみる

TerraformでCloudWatchのログ保管期間だけ変えてみる。

```terraform
terraform {
  required_providers {
    aws = {
      source = "hashicorp/aws"
      version = "~> 6.0"
    }
  }
}

provider "aws" {
  region = "ap-northeast-1"
}

resource "aws_cloudwatch_log_group" "practice" {
  name = "cloudwatch_log_group"
  retention_in_days = 14

}
```
